\providecolor{benchAxis}{HTML}{222222}
\providecolor{benchGrid}{HTML}{D8DDE3}
\providecolor{benchTorch}{HTML}{C9D3DF}
\providecolor{benchNpu}{HTML}{2F3A40}

\begin{tikzpicture}[
    x=1cm,
    y=1cm,
    axis/.style={benchAxis,line width=0.55pt},
    grid/.style={benchGrid,line width=0.35pt},
    tick/.style={font=\footnotesize,text=benchAxis},
    ylabel/.style={font=\footnotesize,text=benchAxis,align=right},
    value/.style={font=\footnotesize,text=benchAxis},
]
\def\xorigin{2.35}
\def\yorigin{0.35}
\def\xscale{0.052}
\def\xmax{170}
\def\barheight{0.22}
\def\ytorch{1.80}
\def\ynpu{0.92}

\pgfmathsetmacro{\xright}{\xorigin+\xmax*\xscale}
\pgfmathsetmacro{\xlabel}{\xorigin+85*\xscale}
\pgfmathsetmacro{\xtorch}{\xorigin+155.93*\xscale}
\pgfmathsetmacro{\xnpu}{\xorigin+5.56*\xscale}
\pgfmathsetmacro{\xtorchstdleft}{\xorigin+(155.93-5.17)*\xscale}
\pgfmathsetmacro{\xtorchstdright}{\xorigin+(155.93+5.17)*\xscale}
\pgfmathsetmacro{\xnpustdleft}{\xorigin+(5.56-1.71)*\xscale}
\pgfmathsetmacro{\xnpustdright}{\xorigin+(5.56+1.71)*\xscale}

\path[use as bounding box] (0.00,-0.25) rectangle (11.25,2.25);

% Axis and grid
\foreach \t in {0,40,80,120,160} {
    \pgfmathsetmacro{\xtick}{\xorigin+\t*\xscale}
    \draw[grid] (\xtick,\yorigin) -- (\xtick,2.08);
    \draw[axis] (\xtick,\yorigin) -- (\xtick,\yorigin-0.05);
    \node[tick,anchor=north] at (\xtick,\yorigin-0.11) {\t};
}
\draw[axis] (\xorigin,\yorigin) -- (\xright,\yorigin);
\draw[axis] (\xorigin,\yorigin) -- (\xorigin,2.08);
\node[tick,anchor=north] at (\xlabel,\yorigin-0.43) {Latency (ms)};

% Bars
\filldraw[draw=benchAxis,fill=benchTorch,line width=0.45pt]
    (\xorigin,\ytorch-\barheight) rectangle (\xtorch,\ytorch+\barheight);
\filldraw[draw=benchAxis,fill=benchNpu,line width=0.45pt]
    (\xorigin,\ynpu-\barheight) rectangle (\xnpu,\ynpu+\barheight);

% Standard deviation bars
\draw[axis] (\xtorchstdleft,\ytorch) -- (\xtorchstdright,\ytorch);
\draw[axis] (\xtorchstdleft,\ytorch-0.08) -- (\xtorchstdleft,\ytorch+0.08);
\draw[axis] (\xtorchstdright,\ytorch-0.08) -- (\xtorchstdright,\ytorch+0.08);
\draw[axis] (\xnpustdleft,\ynpu) -- (\xnpustdright,\ynpu);
\draw[axis] (\xnpustdleft,\ynpu-0.08) -- (\xnpustdleft,\ynpu+0.08);
\draw[axis] (\xnpustdright,\ynpu-0.08) -- (\xnpustdright,\ynpu+0.08);

% Labels and values
\node[ylabel,anchor=east] at (\xorigin-0.25,\ytorch) {torch};
\node[ylabel,anchor=east] at (\xorigin-0.25,\ynpu) {NPU fused};
\node[value,anchor=west] at (\xorigin+0.35,\ytorch)
    {155.93 $\pm$ 5.17 ms};
\node[value,anchor=west] at (\xorigin+0.45,\ynpu)
    {5.56 $\pm$ 1.71 ms};

\end{tikzpicture}
